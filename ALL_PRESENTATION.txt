package vn.id.quiz.quizcitbackend.modules.content.presentation.dto;

import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.model.Quiz;

/**
 * Phase 2 (P1) - Response rút gọn cho {@code Quiz}.
 *
 * <p>Được trả về khi gọi {@code GET /api/subjects/{subjectId}/quizzes} để
 * liệt kê các đề thi thuộc một môn học mà không cần đính kèm toàn bộ câu hỏi.</p>
 */
@Getter
@Setter
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class QuizSummaryResponse {
    private String quizId;
    private String name;
    private String subjectId;
    private String chapterId;
    private Integer totalQuestion;
    private Integer duration;
    private String type;

    public static QuizSummaryResponse from(Quiz quiz) {
        if (quiz == null) {
            return null;
        }
        return QuizSummaryResponse.builder()
                .quizId(quiz.getQuizId())
                .name(quiz.getName())
                .subjectId(quiz.getSubjectId())
                .chapterId(quiz.getChapterId())
                .totalQuestion(quiz.getTotalQuestion())
                .duration(quiz.getDuration())
                .type(quiz.getType() != null ? quiz.getType().name() : null)
                .build();
    }
}

package vn.id.quiz.quizcitbackend.modules.content.presentation.model;
import jakarta.persistence.*;
import lombok.*;

import java.time.ZonedDateTime;
import java.util.ArrayList;
import java.util.List;
import java.util.UUID;

@Entity
@Table(name = "subjects")
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@AllArgsConstructor
@Builder
public class Subject {
    @Id
    @Column(name = "subject_id", length = 50) // cơ bản mặc định length của UUID là 36 nhưng để vầy đề phòng đổi bên cung cấp UUID
    private String subjectId;

    @Column(name = "course_code", unique = true, nullable = false, length = 20)
    private String courseCode;

    @Column(nullable = false, length = 200)
    private String name;

    @Column(columnDefinition = "TEXT")
    private String description;

    @Column(name = "created_at")
    private ZonedDateTime createdAt;

    @OneToMany(mappedBy = "subject", cascade = CascadeType.ALL, orphanRemoval = true) //map với field subject trong chapter

    public static Subject create(String courseCode, String name, String description){
        if(courseCode==null||courseCode.isBlank()){
            throw new IllegalArgumentException("Mã học phần không được để trống");
        }
        if(name==null||name.isBlank()){
            throw new IllegalArgumentException("Tên học phần không được để trống");
        }
        return Subject.builder()
                .subjectId(UUID.randomUUID().toString()) // Chủ động sinh ID
                .courseCode(courseCode)
                .name(name)
                .description(description)
                .createdAt(ZonedDateTime.now())
                .build();
    }
    public void updateSubject(String courseCode, String name, String description){
        this.courseCode = courseCode;
        this.name = name;
        this.description = description;
    }

}

    // Bang user_achievement dung khoa chinh ghep user_id + achieve_id.
    @Id
    @Column(name = "user_id", length = 50)
    private String userId;

    @Id
    @Column(name = "achieve_id", length = 50)

    @Column(name = "is_achieved")
    private Boolean achieved;

    @Column(length = 100)
    private String badge;

    @Column(columnDefinition = "TEXT")
    private String description;

    @Column(name = "condition_type", length = 50)
    private String conditionType;

    @Column(name = "condition_value")
    private Integer conditionValue;

    @Column(name = "achieved_at")
    private ZonedDateTime achievedAt;

    // Tao thanh tuu chua dat; khi user dat dieu kien se goi markAchieved().
    public static UserAchievement create(
            String userId,
            String achieveId,
            String badge,
            String description,
            String conditionType,
            Integer conditionValue) {
        validateId(userId, "Nguoi dung khong hop le");
        validateId(achieveId, "Thanh tuu khong hop le");

        return UserAchievement.builder()
                .userId(userId)
                .achieveId(achieveId)
                .achieved(false)
                .badge(badge)
                .description(description)
                .conditionType(conditionType)
                .conditionValue(conditionValue)
                .build();
    }

    // Danh dau user da dat thanh tuu va luu thoi diem dat.
    public void markAchieved() {
        this.achieved = true;
        this.achievedAt = ZonedDateTime.now();
    }

    // Reset trang thai thanh tuu neu can tinh lai dieu kien.
    public void markUnachieved() {
        this.achieved = false;
        this.achievedAt = null;
    }

    private static void validateId(String value, String message) {
        if (value == null || value.isBlank()) {
            throw new AppException(message, HttpStatus.BAD_REQUEST);
        }
    }
}

package vn.id.quiz.quizcitbackend.modules.user.presentation.model;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import lombok.AccessLevel;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

import java.time.ZonedDateTime;
import java.util.UUID;

@Entity
@Table(name = "users")
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@AllArgsConstructor
@Builder
public class User {
    @Id
    // Khoa chinh cua bang users, tu sinh bang UUID dang String.
    @Column(name = "user_id", length = 50)
    private String userId;

    @Setter
    @Column(nullable = false, unique = true, length = 50)
    private String username;

    @Setter
    @Column(nullable = false, length = 255)
    private String password;

    @Setter
    @Column(nullable = false, unique = true, length = 100)
    private String email;

    @Setter
    @Column(name = "display_name", length = 100)
    private String displayName;

    @Setter
    @Enumerated(EnumType.STRING)
    @Column(length = 20, columnDefinition = "user_status_enum")
    private UserStatus status;

    @Column(name = "created_at")
    private ZonedDateTime createdAt;


    // Factory chi tao entity theo schema users va gan gia tri mac dinh.
    // Validation va nghiep vu chinh nam o service.
    public static User create(String username, String password, String email, String displayName) {
        ZonedDateTime now = ZonedDateTime.now();

        return User.builder()
                .userId(UUID.randomUUID().toString())
                .username(username.trim())
                .password(password)
                .email(email.trim())
                .displayName(normalizeDisplayName(displayName, username))
                .status(UserStatus.UNVERIFIED)
                .createdAt(now)
                .build();
    }

    private static String normalizeDisplayName(String displayName, String username) {
        if (displayName == null || displayName.isBlank()) {
            return username.trim();
        }
        return displayName.trim();
    }
}
