package vn.id.quiz.quizcitbackend.core.module;


/**
 * Interface dành cho các module của App, bắt buộc các module của app phải implements interface này
 */
public interface AppModule {
    /**
     * Phương thức dùng để khai báo bản thân module với app
     * <br/>Thực chất DI đã tự tim nó vào app rồi, nhưng vấn đề là nó sẽ ngủ cho đến khi được gọi hàm
     * <br/>Việc gọi initialize là một cách chủ động để module chạy ngay, không rơi vào trạng thái chừo
     * */
    public void initialize();
    /**
     * Dùng để lấy name cho ModuleEngine
     * */
    public String getName();
}

package vn.id.quiz.quizcitbackend.core.module;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.context.event.ApplicationReadyEvent;
import org.springframework.context.event.EventListener;
import org.springframework.stereotype.Component;

import java.sql.SQLOutput;
import java.util.List;

@Component
public class ModuleEngine {
    private final List<AppModule> availableModules;

    //Thật ra trong Spring 4.3 trở đi thì không cần autowire các component chỉ có duy nhất một constructor
    @Autowired // Vẫn thêm Autowired cho dễ đọc
    public ModuleEngine(List<AppModule> availableModules){
        this.availableModules = availableModules;
    }
//Cái event bên dưới tự được spring publish sau khi khởi tạo xong các thành phần.
    @EventListener(ApplicationReadyEvent.class)
    public void runModules(){
        System.out.println("==== Đang bắt đầu kích hoạt các module ====");
        availableModules.forEach(module->{
            System.out.println("- Đang gọi initialize() của module: " + module.getName());
            try{
                module.initialize();
            }catch(Exception e){ //Nhớ đổi kiểu exception nếu có kiểu custom cho lỗi này
                System.out.println("!!! Gặp lỗi khi kích hoạt module "+module.getName()+" !!!");
                System.out.println("---- Start traceback lỗi ----");
                e.printStackTrace();
                System.out.println("---- End traceback lỗi ----");
            }
        });
        System.out.println("==== Tất cả module đã được nạp thành công ====");

    }
}

package vn.id.quiz.quizcitbackend.modules.ai.client;

import org.springframework.stereotype.Component;

import lombok.RequiredArgsConstructor;
import vn.id.quiz.quizcitbackend.modules.ai.dto.EstimateThetaRequest;
import vn.id.quiz.quizcitbackend.modules.ai.dto.EstimateThetaResponse;
import org.springframework.web.client.RestClient;

@Component
@RequiredArgsConstructor
public class AIClient {

    private final RestClient aiRestClient;

    public double estimateTheta(
            EstimateThetaRequest request) {

        EstimateThetaResponse response = aiRestClient.post()
                .uri("https://quizcit-ai.onrender.com")
                .body(request)
                .retrieve()
                .body(EstimateThetaResponse.class);

        return response.theta();
    }
}

package vn.id.quiz.quizcitbackend.modules.auth.service;

import org.springframework.stereotype.Service;

import javax.crypto.SecretKeyFactory;
import javax.crypto.spec.PBEKeySpec;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.SecureRandom;
import java.security.spec.KeySpec;
import java.util.Base64;

@Service
public class PasswordService {
    private static final String HASH_PREFIX = "pbkdf2";
    private static final String HASH_SEPARATOR = "$";
    private static final String HASH_ALGORITHM = "PBKDF2WithHmacSHA256";
    private static final int ITERATIONS = 185_000;
    private static final int SALT_LENGTH_BYTES = 16;
    private static final int KEY_LENGTH_BITS = 256;

    private final SecureRandom secureRandom = new SecureRandom();

    public String hash(String rawPassword) {
        validateRawPassword(rawPassword);

        byte[] salt = new byte[SALT_LENGTH_BYTES];
        secureRandom.nextBytes(salt);
        byte[] hash = pbkdf2(rawPassword.toCharArray(), salt, ITERATIONS);

        return String.join(
                HASH_SEPARATOR,
                HASH_PREFIX,
                String.valueOf(ITERATIONS),
                Base64.getEncoder().encodeToString(salt),
                Base64.getEncoder().encodeToString(hash)
        );
    }

    public boolean matches(String rawPassword, String storedPassword) {
        if (rawPassword == null || storedPassword == null || storedPassword.isBlank()) {
            return false;
        }

        if (!isHashed(storedPassword)) {
            return constantTimeEquals(rawPassword.getBytes(StandardCharsets.UTF_8), storedPassword.getBytes(StandardCharsets.UTF_8));
        }

        String[] parts = storedPassword.split("\\$");
        if (parts.length != 4) {
            return false;
        }

        try {
            int iterations = Integer.parseInt(parts[1]);
            byte[] salt = Base64.getDecoder().decode(parts[2]);
            byte[] expectedHash = Base64.getDecoder().decode(parts[3]);
            byte[] actualHash = pbkdf2(rawPassword.toCharArray(), salt, iterations);
            return constantTimeEquals(actualHash, expectedHash);
        } catch (IllegalArgumentException ex) {
            return false;
        }
    }

    public boolean isHashed(String storedPassword) {
        return storedPassword != null && storedPassword.startsWith(HASH_PREFIX + HASH_SEPARATOR);
    }

    private byte[] pbkdf2(char[] password, byte[] salt, int iterations) {
        try {
            KeySpec spec = new PBEKeySpec(password, salt, iterations, KEY_LENGTH_BITS);
            SecretKeyFactory factory = SecretKeyFactory.getInstance(HASH_ALGORITHM);
            return factory.generateSecret(spec).getEncoded();
        } catch (Exception ex) {
            throw new IllegalStateException("Khong the ma hoa mat khau", ex);
        }
    }

    private boolean constantTimeEquals(byte[] actual, byte[] expected) {
        return MessageDigest.isEqual(actual, expected);
    }

    private void validateRawPassword(String rawPassword) {
        if (rawPassword == null || rawPassword.isBlank()) {
            throw new IllegalArgumentException("Mat khau khong duoc de trong");
        }
    }
}

package vn.id.quiz.quizcitbackend.modules.auth.service;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import vn.id.quiz.quizcitbackend.modules.user.presentation.model.User;
import vn.id.quiz.quizcitbackend.modules.user.presentation.model.UserRole;

import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.time.Instant;
import java.time.ZoneId;
import java.time.ZonedDateTime;
import java.util.Base64;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

@Service
public class TokenService {
    private static final String TOKEN_TYPE = "Bearer";
    private static final String HMAC_ALGORITHM = "HmacSHA256";
    private static final String JWT_HEADER = "{\"alg\":\"HS256\",\"typ\":\"JWT\"}";
    private static final Pattern EXPIRATION_PATTERN = Pattern.compile("\"exp\":(\\d+)");

    private final byte[] secret;
    private final long expirationSeconds;

    public TokenService(
            @Value("${quizcit.auth.token.secret}") String secret,
            @Value("${quizcit.auth.token.expiration-seconds:86400}") long expirationSeconds
    ) {
        if (secret == null || secret.isBlank()) {
            throw new IllegalStateException("quizcit.auth.token.secret chua duoc cau hinh");
        }
        this.secret = secret.getBytes(StandardCharsets.UTF_8);
        this.expirationSeconds = expirationSeconds;
    }

    public TokenDetails generate(User user, UserRole role) {
        Instant issuedAt = Instant.now();
        Instant expiresAt = issuedAt.plusSeconds(expirationSeconds);

        String encodedHeader = base64Url(JWT_HEADER.getBytes(StandardCharsets.UTF_8));
        String encodedPayload = base64Url(buildPayload(user, role, issuedAt, expiresAt).getBytes(StandardCharsets.UTF_8));
        String unsignedToken = encodedHeader + "." + encodedPayload;
        String signature = base64Url(sign(unsignedToken));

        return new TokenDetails(
                unsignedToken + "." + signature,
                TOKEN_TYPE,
                ZonedDateTime.ofInstant(issuedAt, ZoneId.systemDefault()),
                ZonedDateTime.ofInstant(expiresAt, ZoneId.systemDefault())
        );
    }

    public boolean isValid(String token) {
        if (token == null || token.isBlank()) {
            return false;
        }

        String[] parts = token.split("\\.");
        if (parts.length != 3) {
            return false;
        }

        String unsignedToken = parts[0] + "." + parts[1];
        String expectedSignature = base64Url(sign(unsignedToken));
        if (!MessageDigest.isEqual(
                expectedSignature.getBytes(StandardCharsets.UTF_8),
                parts[2].getBytes(StandardCharsets.UTF_8)
        )) {
            return false;
        }

        return extractExpiration(parts[1])
                .map(expiration -> expiration >= Instant.now().getEpochSecond())
                .orElse(false);
    }

    private String buildPayload(User user, UserRole role, Instant issuedAt, Instant expiresAt) {
        String roleName = role == null ? null : role.name();
        String status = user.getStatus() == null ? null : user.getStatus().name();

        return "{"
                + "\"sub\":" + quote(user.getUserId()) + ","
                + "\"username\":" + quote(user.getUsername()) + ","
                + "\"email\":" + quote(user.getEmail()) + ","
                + "\"role\":" + quote(roleName) + ","
                + "\"status\":" + quote(status) + ","
                + "\"iat\":" + issuedAt.getEpochSecond() + ","
                + "\"exp\":" + expiresAt.getEpochSecond()
                + "}";
    }

    private byte[] sign(String content) {
        try {
            Mac mac = Mac.getInstance(HMAC_ALGORITHM);
            mac.init(new SecretKeySpec(secret, HMAC_ALGORITHM));
            return mac.doFinal(content.getBytes(StandardCharsets.UTF_8));
        } catch (Exception ex) {
            throw new IllegalStateException("Khong the ky token", ex);
        }
    }

    private java.util.Optional<Long> extractExpiration(String encodedPayload) {
        try {
            String payload = new String(Base64.getUrlDecoder().decode(encodedPayload), StandardCharsets.UTF_8);
            Matcher matcher = EXPIRATION_PATTERN.matcher(payload);
            if (!matcher.find()) {
                return java.util.Optional.empty();
            }
            return java.util.Optional.of(Long.parseLong(matcher.group(1)));
        } catch (IllegalArgumentException ex) {
            return java.util.Optional.empty();
        }
    }

    private String base64Url(byte[] value) {
        return Base64.getUrlEncoder().withoutPadding().encodeToString(value);
    }

    private String quote(String value) {
        if (value == null) {
            return "null";
        }
        return "\"" + value
                .replace("\\", "\\\\")
                .replace("\"", "\\\"")
                + "\"";
    }
}

package vn.id.quiz.quizcitbackend.modules.auth.service;

import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import vn.id.quiz.quizcitbackend.modules.auth.presentation.dto.AuthResponse;
import vn.id.quiz.quizcitbackend.modules.auth.presentation.dto.LoginRequest;
import vn.id.quiz.quizcitbackend.modules.auth.presentation.dto.RegisterRequest;
import vn.id.quiz.quizcitbackend.modules.auth.repository.AuthRepository;
import vn.id.quiz.quizcitbackend.modules.user.presentation.model.Role;
import vn.id.quiz.quizcitbackend.modules.user.presentation.model.User;
import vn.id.quiz.quizcitbackend.modules.user.presentation.model.UserRole;
import vn.id.quiz.quizcitbackend.modules.user.presentation.model.UserRoleAssignment;
import vn.id.quiz.quizcitbackend.modules.user.presentation.model.UserStatus;
import vn.id.quiz.quizcitbackend.shared.exceptions.AppException;

import java.time.ZonedDateTime;
import java.util.Optional;

@Service
@RequiredArgsConstructor
public class AuthServiceImpl implements AuthUseCase {
    private final AuthRepository authRepository;
    private final AuthMapper authMapper;
    private final PasswordService passwordService;
    private final TokenService tokenService;

    @Override
    @Transactional
    public AuthResponse register(RegisterRequest request) {
        if (authRepository.existsByUsername(request.username())) {
            throw new AppException("Ten dang nhap da ton tai", HttpStatus.CONFLICT);
        }
        if (authRepository.existsByEmail(request.email())) {
            throw new AppException("Email da ton tai", HttpStatus.CONFLICT);
        }

        String hashedPassword = passwordService.hash(request.password());
        User user = User.create(request.username(), hashedPassword, request.email(), request.displayName());
        User savedUser = authRepository.save(user);

        Role defaultRole = getOrCreateRole(UserRole.USER);
        authRepository.save(UserRoleAssignment.assign(savedUser.getUserId(), defaultRole.getRoleId()));

        TokenDetails tokenDetails = tokenService.generate(savedUser, defaultRole.getName());
        return authMapper.toAuthResponse(savedUser, defaultRole.getName(), tokenDetails);
    }

    @Override
    @Transactional
    public AuthResponse login(LoginRequest request) {
        User user = findUserByIdentifier(request.identifier())
                .orElseThrow(this::invalidCredentials);

        ensureUserCanAuthenticate(user);
        ensurePasswordMatches(request.password(), user.getPassword());
        upgradeLegacyPasswordIfNeeded(user, request.password());

        UserRole role = resolveUserRole(user.getUserId());
        TokenDetails tokenDetails = tokenService.generate(user, role);
        return authMapper.toAuthResponse(user, role, tokenDetails);
    }

    private Optional<User> findUserByIdentifier(String identifier) {
        if (identifier == null || identifier.isBlank()) {
            return Optional.empty();
        }

        String normalizedIdentifier = identifier.trim();
        return authRepository.findUserByUsername(normalizedIdentifier)
                .or(() -> authRepository.findUserByEmail(normalizedIdentifier));
    }

    private void ensureUserCanAuthenticate(User user) {
        if (user.getStatus() == UserStatus.BANNED) {
            throw new AppException("Tai khoan da bi khoa", HttpStatus.FORBIDDEN);
        }
    }

    private void ensurePasswordMatches(String rawPassword, String storedPassword) {
        if (!passwordService.matches(rawPassword, storedPassword)) {
            throw invalidCredentials();
        }
    }

    private void upgradeLegacyPasswordIfNeeded(User user, String rawPassword) {
        if (!passwordService.isHashed(user.getPassword())) {
            user.setPassword(passwordService.hash(rawPassword));
            user.setUpdatedAt(ZonedDateTime.now());
            authRepository.save(user);
        }
    }

    private Role getOrCreateRole(UserRole roleName) {
        return authRepository.findRoleByName(roleName)
                .orElseGet(() -> authRepository.save(Role.create(roleName)));
    }

    private UserRole resolveUserRole(String userId) {
        return authRepository.findRoleByUserId(userId)
                .map(Role::getName)
                .orElse(null);
    }

    private AppException invalidCredentials() {
        return new AppException("Thong tin dang nhap khong chinh xac", HttpStatus.UNAUTHORIZED);
    }
}

package vn.id.quiz.quizcitbackend.modules.content.presentation.model;
import jakarta.persistence.*;
import lombok.*;

import java.time.ZonedDateTime;
import java.util.ArrayList;
import java.util.List;
import java.util.UUID;

@Entity
@Table(name = "subjects")
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@AllArgsConstructor
@Builder
public class Subject {
    @Id
    @Column(name = "subject_id", length = 50) // cơ bản mặc định length của UUID là 36 nhưng để vầy đề phòng đổi bên cung cấp UUID
    private String subjectId;

    @Column(name = "course_code", unique = true, nullable = false, length = 20)
    private String courseCode;

    @Column(nullable = false, length = 200)
    private String name;

    @Column(columnDefinition = "TEXT")
    private String description;

    @Column(name = "created_at")
    private ZonedDateTime createdAt;

    @OneToMany(mappedBy = "subject", cascade = CascadeType.ALL, orphanRemoval = true) //map với field subject trong chapter

    public static Subject create(String courseCode, String name, String description){
        if(courseCode==null||courseCode.isBlank()){
            throw new IllegalArgumentException("Mã học phần không được để trống");
        }
        if(name==null||name.isBlank()){
            throw new IllegalArgumentException("Tên học phần không được để trống");
        }
        return Subject.builder()
                .subjectId(UUID.randomUUID().toString()) // Chủ động sinh ID
                .courseCode(courseCode)
                .name(name)
                .description(description)
                .createdAt(ZonedDateTime.now())
                .build();
    }
    public void updateSubject(String courseCode, String name, String description){
        this.courseCode = courseCode;
        this.name = name;
        this.description = description;
    }

}

package vn.id.quiz.quizcitbackend.modules.content.service;

import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Chapter;
import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Subject;
import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Theory;
import vn.id.quiz.quizcitbackend.modules.content.presentation.dto.*;
import vn.id.quiz.quizcitbackend.modules.content.repository.ContentRepository;
import vn.id.quiz.quizcitbackend.shared.exceptions.AppException;

import java.util.List;

@Service
@RequiredArgsConstructor
public class ContentServiceImplement implements ContentService {
    private final ContentRepository contentRepository;
    private final ContentMapper mapper;

    @Override
    @Transactional(readOnly = true)
    public List<SubjectResponse> getAllSubject(){
        List<Subject> subjects = contentRepository.findAllSubject();
        return mapper.toSubjectResponseList(subjects);
    }
    @Override
    @Transactional(readOnly = true)
    public SubjectResponse getSubjectById(String subjectId){

        Subject subject = contentRepository.findSubjectById(subjectId)
                .orElseThrow(()-> new AppException("Không tìm thấy học phần được yêu cầu", HttpStatus.NOT_FOUND));
        return mapper.toSubjectResponse(subject);
    }
    @Override
    @Transactional(readOnly = true)
    public ChapterResponse getChapterById(String chapterId){
        Chapter chapter = contentRepository.findChapterById(chapterId)
                .orElseThrow(()-> new AppException("Không tìm thấy chương được yêu cầu", HttpStatus.NOT_FOUND));
        return mapper.toChapterResponse(chapter);
    }
    @Override
    @Transactional(readOnly = true)
    public TheoryResponse getTheoryById(String theoryId){
        Theory theory = contentRepository.findTheoryById(theoryId)
                .orElseThrow(()-> new AppException("Không tìm thấy nội dung lý thuyết được yêu cầu", HttpStatus.NOT_FOUND));
        return mapper.toTheoryResponse(theory);
    }

    @Override
    @Transactional
    public SubjectResponse createSubject(CreateSubjectRequest request){

        if(contentRepository.existsByCourseCode(request.courseCode())){
            throw new AppException("Mã học phần đã tồn tại.", HttpStatus.CONFLICT);
        }
        Subject subject = Subject.create(request.courseCode(), request.name(), request.description());
        return mapper.toSubjectResponse(contentRepository.save(subject));
    }

    @Override
    @Transactional
    public ChapterResponse addChapter(String subjectId, CreateChapterRequest request){
        if(!contentRepository.existsBySubjectId(subjectId)){
            throw new AppException("Không tìm thấy học phần được yêu cầu", HttpStatus.NOT_FOUND);
        }
        Chapter response = Chapter.create(subjectId, request.name());
        contentRepository.save(response);
        return mapper.toChapterResponse(response);
    }
    @Override
    @Transactional
    public TheoryResponse addTheory(String chapterId, CreateTheoryRequest request){
        if(!contentRepository.existsByChapterId(chapterId)){

    private final ContentRepository contentRepository;

    public QuestionResponse createQuestion(
            CreateQuestionRequest request) {

        Subject subject = contentRepository.findSubjectById(
                request.getSubjectId())
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy môn học",
                        HttpStatus.NOT_FOUND));

        Chapter chapter = contentRepository.findChapterById(

        for (var opt : request.getOptions()) {

            Option option = Option.create(
                    question.getQuestionId(),
                    opt.getContent(),
                    opt.isCorrected());

            quizRepository.save(option);
        }

        return mapper.toResponse(question);

                .subject(question.getSubject())

                .explaination(request.getExplaination())
                .difficultyB(request.getDifficultyB())

                .isActive(question.isActive())

package vn.id.quiz.quizcitbackend.modules.quiz.service;

import java.util.Collections;
import java.util.List;

import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import lombok.RequiredArgsConstructor;

import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Chapter;
import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Subject;
import vn.id.quiz.quizcitbackend.modules.content.repository.ContentRepository;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.dto.CreateQuizRequest;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.dto.QuizResponse;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.dto.UpdateQuizRequest;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.model.Question;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.model.Quiz;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.model.QuizQuestion;
import vn.id.quiz.quizcitbackend.modules.quiz.repository.QuizRepository;
import vn.id.quiz.quizcitbackend.shared.exceptions.AppException;

@Service
@RequiredArgsConstructor
public class QuizServiceImplement implements QuizService {

    private final QuizRepository quizRepository;
    private final ContentRepository contentRepository;
    private final QuizMapper mapper;

    @Override
    @Transactional
    public QuizResponse createQuiz(
            CreateQuizRequest request) {

        Subject subject = contentRepository.findSubjectById(
                request.getSubjectId())
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy môn học",
                        HttpStatus.NOT_FOUND));

        Chapter chapter = contentRepository.findChapterById(
                request.getChapterId())
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy chương học",
                        HttpStatus.NOT_FOUND));

        List<Question> questions = quizRepository.findQuestionsByChapterId(
                chapter.getChapterId());

        if (questions.isEmpty()) {
            throw new AppException(
                    "Không có câu hỏi",
                    HttpStatus.BAD_REQUEST);
        }

        Collections.shuffle(questions);

        List<Question> selected = questions.stream()
                .limit(request.getTotalQuestion())
                .toList();

        Quiz quiz = Quiz.create(
                subject,
                chapter,
                request.getName(),
                selected.size(),
                request.getType(),
                request.getDuration());

        quizRepository.save(quiz);

        int order = 1;

        for (Question q : selected) {

            quizRepository.save(
                    QuizQuestion.builder()
                            .quiz(quiz)
                            .question(q)
                            .indexOrder(order++)
                            .build());
        }

        return mapper.toResponse(quiz);
    }

    @Override
    @Transactional
    public void updateQuiz(
            String quizId,
            UpdateQuizRequest request) {

        Quiz quiz = quizRepository.findQuizById(
                quizId)
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy quiz",
                        HttpStatus.NOT_FOUND));

        Quiz updated = Quiz.builder()
                .quizId(quiz.getQuizId())
                .subject(quiz.getSubject())
                .chapter(quiz.getChapter())
                .name(request.getName())
                .type(quiz.getType())
                .totalQuestion(quiz.getTotalQuestion())
                .duration(request.getDuration())
                .build();

        quizRepository.save(updated);
    }

    @Override
    @Transactional
    public void deleteQuiz(String quizId) {

        quizRepository.deleteQuizById(quizId);
        quizRepository.deleteQuizQuestionsByQuizId(quizId);
    }

    @Override
    public QuizResponse getQuizById(String quizId) {

        return mapper.toResponse(
                quizRepository.findQuizById(quizId)
                        .orElseThrow(() -> new AppException(
                                "Không tìm thấy quiz",
                                HttpStatus.NOT_FOUND)));
    }

    @Override
    public List<QuizResponse> getQuizzesByChapter(
            String chapterId) {

        return mapper.toResponseList(
                quizRepository.findQuizzesByChapterId(chapterId));
    }

package vn.id.quiz.quizcitbackend.modules.quiz.service;

import java.util.HashMap;
import java.util.List;
import java.util.Map;

import org.springframework.http.HttpStatus;

import lombok.RequiredArgsConstructor;
import vn.id.quiz.quizcitbackend.modules.ai.dto.EstimateThetaRequest;
import vn.id.quiz.quizcitbackend.modules.ai.dto.EstimateThetaResponse;
import vn.id.quiz.quizcitbackend.modules.ai.service.AIService;

import vn.id.quiz.quizcitbackend.modules.quiz.repository.QuizRepository;
import vn.id.quiz.quizcitbackend.shared.exceptions.AppException;

@Service
@RequiredArgsConstructor
public class AttemptServiceImplement
        implements AttemptService {

    private final QuizRepository quizRepository;
    private final AttemptMapper mapper;
    private final ProgressRepository progressRepository;

    private final AIService aiService;

    @Override
    @Transactional
    public StartQuizResponse startQuiz(
            String userId,
            String quizId) {

        Quiz quiz = quizRepository.findQuizById(quizId)
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy đề thi",
                        HttpStatus.NOT_FOUND));

        List<QuizQuestion> questions = quizRepository.findQuizQuestionsByQuizId(
                quizId);

        if (questions.isEmpty()) {
            throw new AppException(
                    "Đề thi không có câu hỏi",
                    HttpStatus.BAD_REQUEST);
        }

        Attempt attempt = Attempt.create(
                userId,
                quizId,
                questions.size());

        quizRepository.save(attempt);

        return mapper.toStartQuizResponse(attempt);
    }

    @Override
    @Transactional
    public QuizResultResponse submitQuiz(
            SubmitQuizRequest request) {
        // Kiếm lần thử để tạo bài nộp
        Attempt attempt = quizRepository
                .findAttemptById(
                        request.getAttemptId())
                .orElseThrow(
                        () -> new AppException(
                                "Không tìm thấy attempt",
                                HttpStatus.NOT_FOUND));
        // Tạo danh sách các skill có trong bài
        Map<String, SkillThetaData> skillMap = new HashMap<>();

        // Duyệt đáp án
        int correctCount = 0;

        for (SubmitAnswerRequest answer : request.getAnswers()) {
            Question question = quizRepository
                    .findQuestionById(
                            answer.getQuestionId())
                    .orElseThrow();
            Option option = quizRepository
                    .findOptionById(
                            answer.getOptionId())
                    .orElseThrow();
            boolean correct = option.isCorrected();
            Response response = Response.create(
                    attempt.getAttemptId(),
                    question.getQuestionId(),
                    option.getOptionId(),
                    correct);
            quizRepository.save(response);
            if (correct) {
                correctCount++;
            }
            List<QuestionSkill> mappings = progressRepository
                    .findQuestionSkillsByQuestionId(
                            question.getQuestionId());
            for (QuestionSkill qs : mappings) {

                SkillThetaData data = skillMap.computeIfAbsent(
                        qs.getSkillId(),
                        id -> new SkillThetaData());

                data.getResponses()
                        .add(correct ? 1 : 0);

                data.getDifficulties()
                        .add(question.getDifficultyB());

                data.getWeights()
                        .add(1.0);
            }
        }
        Map<String, Double> skillThetaMap = new HashMap<>();

        for (var entry : skillMap.entrySet()) {

            String skillId = entry.getKey();

            SkillThetaData data = entry.getValue();

            double theta = aiService.estimateTheta(
                    data.getResponses(),
                    data.getDifficulties(),
                    data.getWeights());

            skillThetaMap.put(
                    skillId,
                    theta);

            UserSkillAbility ability = progressRepository
                    .findUserSkillAbility(
                            attempt.getUserId(),
                            skillId)
                    .orElse(null);

            double oldTheta = 0.0;

            if (ability == null) {

                ability = UserSkillAbility.create(
                        attempt.getUserId(),
                        skillId);

            } else {

                oldTheta = ability.getTheta();
            }

            ability.updateAbility(
                    theta,
                    0.0);

            progressRepository.save(
                    ability);

            AbilityHistory history = AbilityHistory.create(
                    attempt.getUserId(),
                    skillId,
                    theta,
                    theta - oldTheta,
                    0.0,
                    attempt.getAttemptId());

            progressRepository.save(
                    history);
        }
        double overallTheta = skillThetaMap.values()
                .stream()
                .mapToDouble(Double::doubleValue)
                .average()
                .orElse(0.0);
        double rawScore = (double) correctCount
                / attempt.getTotalQuestion();
        attempt.complete(
                rawScore,
                overallTheta);

        quizRepository.save(
                attempt);
        return mapper.toQuizResultResponse(
                attempt);
    }

    @Override
    @Transactional(readOnly = true)
    public QuizResultResponse getResult(
            String attemptId) {

        Attempt attempt = quizRepository
                .findAttemptById(attemptId)
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy kết quả bài thi",
                        HttpStatus.NOT_FOUND));

        List<AbilityHistory> histories = progressRepository.findAbilityHistoriesByAttemptId(
                attemptId);

        List<SkillResultResponse> skillResults = histories.stream()
                .map(history -> SkillResultResponse.builder()
                        .skillId(history.getSkillId())
                        .theta(history.getTheta())
                        .standardError(
                                history.getStandardError())
                        .build())
                .toList();

        return QuizResultResponse.builder()
                .attemptId(attempt.getAttemptId())
                .quizId(attempt.getQuizId())
                .rawScore(attempt.getRawScore())
                .totalQuestion(attempt.getTotalQuestion())
                .overallTheta(attempt.getOverallTheta())
                .skillResults(skillResults)
                .build();
    }

    @Override
    @Transactional(readOnly = true)
    public List<QuizResultResponse> getUserResults(
            String userId) {

        List<Attempt> attempts = quizRepository.findAttemptsByUserId(
                userId);

        return mapper.toQuizResultResponseList(
                attempts);
    }
}

        validateId(achieveId, "Thanh tuu khong hop le");

        return UserAchievement.builder()
                .userId(userId)
                .achieveId(achieveId)
                .achieved(false)
                .badge(badge)
                .description(description)
                .conditionType(conditionType)
                .conditionValue(conditionValue)
                .build();

        this.achieved = true;
        this.achievedAt = ZonedDateTime.now();

    public void markUnachieved() {
        this.achieved = false;
        this.achievedAt = null;
    }

            throw new AppException(message, HttpStatus.BAD_REQUEST);

    }

    @Column(nullable = false, unique = true, length = 50)

    @Column(nullable = false, unique = true, length = 100)

    @Column(length = 20)
    private UserStatus status;

    @Column(name = "created_at")
    private ZonedDateTime createdAt;

    @Setter
    @Column(name = "updated_at")
    private ZonedDateTime updatedAt;

    // Validation va nghiep vu chinh nam o service.
    public static User create(String username, String password, String email, String displayName) {
        ZonedDateTime now = ZonedDateTime.now();

        return User.builder()
                .userId(UUID.randomUUID().toString())
                .username(username.trim())
                .password(password)
                .email(email.trim())
                .displayName(normalizeDisplayName(displayName, username))
                .status(UserStatus.UNVERIFIED)
                .createdAt(now)
                .updatedAt(now)
                .build();
    }

    private static String normalizeDisplayName(String displayName, String username) {
        if (displayName == null || displayName.isBlank()) {
            return username.trim();
        }
        return displayName.trim();
    }
}

    private final JpaRoleRepository jpaRoleRepository;

    @Override
    public User save(User user) {
        return jpaUserRepository.save(user);
    }

    @Override
    public Role save(Role role) {

    public UserRoleAssignment save(UserRoleAssignment userRoleAssignment) {
        return jpaUserRoleRepository.save(userRoleAssignment);
    }

    @Override
    public Optional<User> findUserById(String userId) {
        return jpaUserRepository.findById(userId);
    }

    @Override
    public Optional<Role> findRoleByName(UserRole name) {
        return jpaRoleRepository.findByName(name);
    }

    @Override
    public Optional<Role> findRoleByUserId(String userId) {
        // user_roles la bang noi nen can lay role_id truoc roi moi doc bang roles.
        return jpaUserRoleRepository.findByUserId(userId)
                .stream()
                .findFirst()
                .flatMap(userRole -> jpaRoleRepository.findById(userRole.getRoleId()));
    }

    @Override
    public boolean existsById(String userId) {
        return jpaUserRepository.existsById(userId);
    }

    @Override
    public boolean existsByUsername(String username) {
        return jpaUserRepository.existsByUsername(username);
    }

    @Override
    public boolean existsByEmail(String email) {
        return jpaUserRepository.existsByEmail(email);
    }

    @Override
    public boolean existsByUsernameAndUserIdNot(String username, String userId) {
        return jpaUserRepository.existsByUsernameAndUserIdNot(username, userId);
    }

    @Override
    public boolean existsByEmailAndUserIdNot(String email, String userId) {
        return jpaUserRepository.existsByEmailAndUserIdNot(email, userId);

    @Override
    public void deleteRolesByUserId(String userId) {

    }

package vn.id.quiz.quizcitbackend.modules.user.service;

import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import vn.id.quiz.quizcitbackend.modules.auth.service.PasswordService;
import vn.id.quiz.quizcitbackend.modules.user.presentation.dto.ChangePasswordRequest;
import vn.id.quiz.quizcitbackend.modules.user.presentation.dto.UpdateUserProfileRequest;
import vn.id.quiz.quizcitbackend.modules.user.presentation.dto.UserResponse;
import vn.id.quiz.quizcitbackend.modules.user.presentation.model.Role;
import vn.id.quiz.quizcitbackend.modules.user.presentation.model.User;
import vn.id.quiz.quizcitbackend.modules.user.presentation.model.UserRole;
import vn.id.quiz.quizcitbackend.modules.user.repository.UserRepository;
import vn.id.quiz.quizcitbackend.shared.exceptions.AppException;

import java.time.ZonedDateTime;
import java.util.List;

@Service
@RequiredArgsConstructor
public class UserServiceImpl implements UserUseCase {
    private static final int PASSWORD_MIN_LENGTH = 6;
    private static final int PASSWORD_MAX_LENGTH = 255;
    private static final int DISPLAY_NAME_MAX_LENGTH = 100;

    private final UserRepository userRepository;
    private final UserMapper mapper;
    private final PasswordService passwordService;

    // Doc danh sach users va gan role hien tai vao DTO cho frontend.
    @Override
    @Transactional(readOnly = true)
    public List<UserResponse> getAllUsers() {
        List<User> users = userRepository.findAll();
        return users.stream()
                .map(user -> mapper.toUserResponse(user, resolveUserRole(user.getUserId())))
                .toList();
    }

    // Lay chi tiet user; neu khong co thi nem AppException de GlobalExceptionHandler xu ly.
    @Override
    @Transactional(readOnly = true)
    public UserResponse getUserById(String userId) {
        User user = getRequiredUser(userId);
        return mapper.toUserResponse(user, resolveUserRole(userId));
    }

    // Chi cap nhat ten hien thi theo tai lieu kien truc.
    @Override
    @Transactional
    public void updateUserProfile(String userId, UpdateUserProfileRequest request) {
        request = requireRequest(request);
        String displayName = validateDisplayName(request.displayName());

        User user = getRequiredUser(userId);
        user.setDisplayName(displayName);
        touch(user);
        userRepository.save(user);
    }

    // Doi mat khau va luu dang hash, cung co che voi module auth.
    @Override
    @Transactional
    public void changePassword(String userId, ChangePasswordRequest request) {
        request = requireRequest(request);
        String password = validatePassword(request.password());

        User user = getRequiredUser(userId);
        user.setPassword(passwordService.hash(password));
        touch(user);
        userRepository.save(user);
    }

    // Xoa user sau khi kiem tra ton tai; adapter se xoa user_roles lien quan truoc.
    @Override
    @Transactional
    public void deleteUser(String userId) {
        if (!userRepository.existsById(userId)) {
            throw new AppException("Khong tim thay nguoi dung duoc yeu cau", HttpStatus.NOT_FOUND);
        }
        userRepository.deleteById(userId);
    }

    // Dung chung cho cac nghiep vu can dam bao user ton tai truoc khi xu ly.
    private User getRequiredUser(String userId) {
        return userRepository.findUserById(userId)
                .orElseThrow(() -> new AppException("Khong tim thay nguoi dung duoc yeu cau", HttpStatus.NOT_FOUND));
    }

    private <T> T requireRequest(T request) {
        if (request == null) {
            throw new AppException("Yeu cau khong hop le", HttpStatus.BAD_REQUEST);
        }
        return request;
    }

    private String validatePassword(String password) {
        String value = requireNotBlank(password, "Mat khau khong duoc de trong");
        if (value.length() < PASSWORD_MIN_LENGTH || value.length() > PASSWORD_MAX_LENGTH) {
            throw new AppException("Mat khau phai tu 6 den 255 ky tu", HttpStatus.BAD_REQUEST);
        }
        return value;
    }

    private String validateDisplayName(String displayName) {
        String normalized = requireNotBlank(displayName, "Ten hien thi khong duoc de trong").trim();
        if (normalized.length() > DISPLAY_NAME_MAX_LENGTH) {
            throw new AppException("Ten hien thi khong duoc vuot qua 100 ky tu", HttpStatus.BAD_REQUEST);
        }
        return normalized;
    }

    private String requireNotBlank(String value, String message) {
        if (value == null || value.isBlank()) {
            throw new AppException(message, HttpStatus.BAD_REQUEST);
        }
        return value;
    }

    private void touch(User user) {
        user.setUpdatedAt(ZonedDateTime.now());
    }

    // Lay role hien tai cua user tu bang noi user_roles.
    private UserRole resolveUserRole(String userId) {
        return userRepository.findRoleByUserId(userId)
                .map(Role::getName)
                .orElse(null);
    }
}

package vn.id.quiz.quizcitbackend.shared.exceptions;

import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import vn.id.quiz.quizcitbackend.shared.utils.ApiResponse;

import java.util.HashMap;
import java.util.Map;

@RestControllerAdvice
public class GlobalExceptionHandler {
    @ExceptionHandler(AppException.class)
    public ResponseEntity<ApiResponse<Object>> handleAppException(AppException ex){
        return ResponseEntity
                .status(ex.getStatus())
                .body(ApiResponse.error(ex.getStatus(), ex.getMessage()));
    }
    //Lỗi liên quan đến validation trong dữ liệu
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiResponse<Map<String, String>>> handleValidationExceptions(MethodArgumentNotValidException ex) {
        Map<String, String> errors = new HashMap<>();
        ex.getBindingResult().getAllErrors().forEach((error) -> {
            String fieldName = ((FieldError) error).getField();
            String errorMessage = error.getDefaultMessage();
            errors.put(fieldName, errorMessage);
        });

        return ResponseEntity
                .status(HttpStatus.BAD_REQUEST)
                .body(new ApiResponse<>(HttpStatus.BAD_REQUEST.value(), "Dữ liệu không hợp lệ", errors));
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiResponse<Object>> handleGeneralException(Exception ex) {
        // Trong thực tế, nên log lỗi ở đây để debug
        ex.printStackTrace();

        return ResponseEntity
                .status(HttpStatus.INTERNAL_SERVER_ERROR)
                .body(ApiResponse.error(HttpStatus.INTERNAL_SERVER_ERROR.value(), "Đã có lỗi hệ thống xảy ra!"));
    }
}

package vn.id.quiz.quizcitbackend.shared.utils;

/**
 * ApiResponse dùng để thống nhất mọi API trả về luôn có cùng một
 * cấu trúc cho frontend dễ dàng xử lý mà không gặp bất cứ rắc rồi nào
 * @param status mã trạng thái trả về của HTTP
 * @param message thông điệp muốn gửi kèm
 * @param data dữ liệu muốn kèm theo trong response
 * @param <T> Kiểu dữ liêu trả về của data
 */
public record ApiResponse<T>(
        int status,
        String message,
        T data
) {
    public static <T> ApiResponse<T> success(T data){
        return new ApiResponse<>(200, "Success", data);
    }
    public static <T> ApiResponse<T> success(String message, T data){
        return new ApiResponse<>(200, message, data);
    }
    public static <T> ApiResponse<T> error(String message){
        return new ApiResponse<>(400, message, null);
    }
    public static <T> ApiResponse<T> error(int code, String message){
        return new ApiResponse<>(code, message, null);
    }
}

import static org.assertj.core.api.Assertions.assertThatThrownBy;

import static org.mockito.Mockito.verify;
import static org.mockito.Mockito.when;

@ExtendWith(MockitoExtension.class)
class AuthServiceImplTest {
    @Mock
    private AuthRepository authRepository;

    private final AuthMapper authMapper = new AuthMapper() {
    };
    private final PasswordService passwordService = new PasswordService();
    private final TokenService tokenService = new TokenService("test-secret-that-is-long-enough-for-hmac", 3600);

    @Test
    void registerShouldHashPasswordAssignDefaultRoleAndReturnToken() {
        Role defaultRole = Role.create(UserRole.USER);
        RegisterRequest request = new RegisterRequest(
                "student01",
                "secret123",
                "student01@example.com",
                "Student One"
        );

        when(authRepository.existsByUsername(request.username())).thenReturn(false);
        when(authRepository.existsByEmail(request.email())).thenReturn(false);
        when(authRepository.findRoleByName(UserRole.USER)).thenReturn(Optional.of(defaultRole));
        when(authRepository.save(any(User.class))).thenAnswer(invocation -> invocation.getArgument(0));
        when(authRepository.save(any(UserRoleAssignment.class))).thenAnswer(invocation -> invocation.getArgument(0));

        AuthResponse response = service().register(request);

        ArgumentCaptor<User> userCaptor = ArgumentCaptor.forClass(User.class);
        verify(authRepository).save(userCaptor.capture());
        assertThat(userCaptor.getValue().getPassword()).startsWith("pbkdf2$");
        assertThat(passwordService.matches("secret123", userCaptor.getValue().getPassword())).isTrue();

        ArgumentCaptor<UserRoleAssignment> assignmentCaptor = ArgumentCaptor.forClass(UserRoleAssignment.class);
        verify(authRepository).save(assignmentCaptor.capture());
        assertThat(assignmentCaptor.getValue().getRoleId()).isEqualTo(defaultRole.getRoleId());

        assertThat(response.accessToken()).isNotBlank();
        assertThat(response.accessToken().split("\\.")).hasSize(3);
        assertThat(response.tokenType()).isEqualTo("Bearer");
        assertThat(response.expiresAt()).isNotNull();
        assertThat(response.role()).isEqualTo(UserRole.USER);
    }

        Role role = Role.create(UserRole.USER);
        User user = User.create("student01", passwordService.hash("secret123"), "student01@example.com", "Student One");

        when(authRepository.findUserByUsername("student01")).thenReturn(Optional.of(user));
        when(authRepository.findRoleByUserId(user.getUserId())).thenReturn(Optional.of(role));

        AuthResponse response = service().login(new LoginRequest("student01", "secret123"));

        assertThat(response.accessToken()).isNotBlank();
        assertThat(response.tokenType()).isEqualTo("Bearer");
        assertThat(response.userId()).isEqualTo(user.getUserId());
        assertThat(response.username()).isEqualTo("student01");
        assertThat(response.role()).isEqualTo(UserRole.USER);
    }

    @Test

                .isInstanceOf(AppException.class)
                .extracting("status")
                .isEqualTo(HttpStatus.FORBIDDEN.value());
    }

    private AuthServiceImpl service() {
        return new AuthServiceImpl(authRepository, authMapper, passwordService, tokenService);
    }

        userRepository = new FakeUserRepository();
        passwordService = new PasswordService();
        service = new UserServiceImpl(userRepository, new UserMapper() {}, passwordService);
    }

    @Test
    void updateUserProfileShouldOnlyChangeDisplayName() {
        User user = User.create("student01", "hashed-password", "student01@example.com", "Student One");
        userRepository.user = user;

        service.updateUserProfile(user.getUserId(), new UpdateUserProfileRequest("  Student Updated  "));

        assertThat(user.getUsername()).isEqualTo("student01");
        assertThat(user.getEmail()).isEqualTo("student01@example.com");
        assertThat(user.getDisplayName()).isEqualTo("Student Updated");
        assertThat(user.getUpdatedAt()).isAfterOrEqualTo(user.getCreatedAt());
        assertThat(userRepository.savedUser).isSameAs(user);
    }

    @Test
    void updateUserProfileShouldRejectBlankDisplayNameBeforeSaving() {
        assertThatThrownBy(() -> service.updateUserProfile("user-id", new UpdateUserProfileRequest(" ")))
                .isInstanceOf(AppException.class)
                .extracting("status")
                .isEqualTo(HttpStatus.BAD_REQUEST.value());

        assertThat(userRepository.savedUser).isNull();
    }

    @Test
    void changePasswordShouldHashAndPersistNewPassword() {
        User user = User.create("student01", "old-hashed-password", "student01@example.com", "Student One");
        userRepository.user = user;

        service.changePassword(user.getUserId(), new ChangePasswordRequest("newSecret123"));

        assertThat(user.getPassword()).startsWith("pbkdf2$");
        assertThat(user.getPassword()).isNotEqualTo("newSecret123");
        assertThat(passwordService.matches("newSecret123", user.getPassword())).isTrue();

        assertThat(userRepository.savedUser).isSameAs(user);

                .isInstanceOf(AppException.class)
                .extracting("status")
                .isEqualTo(HttpStatus.BAD_REQUEST.value());

        assertThat(userRepository.savedUser).isNull();
    }

        assertThat(userRepository.deletedUserId).isEqualTo("user-id");
    }

        assertThat(userRepository.deletedUserId).isNull();
