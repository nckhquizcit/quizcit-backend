import lombok.RequiredArgsConstructor;

import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Chapter;

import vn.id.quiz.quizcitbackend.modules.content.repository.ContentRepository;

import vn.id.quiz.quizcitbackend.shared.exceptions.AppException;

@Service
@RequiredArgsConstructor
public class QuestionServiceImplement implements QuestionService {

    private final QuizRepository quizRepository;
    private final ContentRepository contentRepository;
    private final QuestionMapper mapper;

    @Override
    @Transactional
    public QuestionResponse createQuestion(
            CreateQuestionRequest request) {

        Subject subject = contentRepository.findSubjectById(
                request.getSubjectId())
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy môn học",
                        HttpStatus.NOT_FOUND));

        Chapter chapter = contentRepository.findChapterById(
                request.getChapterId())
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy chương học",
                        HttpStatus.NOT_FOUND));

        Question question = Question.create(
                subject,
                chapter,
                request.getContent(),
                request.getExplaination(),
                request.getDifficultyB());

        quizRepository.save(question);

        for (var opt : request.getOptions()) {

            Option option = Option.create(
                    question.getQuestionId(),
                    opt.getContent(),
                    opt.isCorrected());

            quizRepository.save(option);
        }

        return mapper.toResponse(question);
    }

    @Override
    @Transactional
    public void updateQuestion(
            String questionId,
            UpdateQuestionRequest request) {

        Question question = quizRepository.findQuestionById(
                questionId)
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy câu hỏi",
                        HttpStatus.NOT_FOUND));

        question = Question.builder()
                .questionId(question.getQuestionId())
                .subject(question.getSubject())
                .chapter(question.getChapter())
                .content(request.getContent())
                .explaination(request.getExplaination())
                .difficultyB(request.getDifficultyB())
                .createdAt(question.getCreatedAt())
                .isActive(question.isActive())
                .build();

        quizRepository.save(question);
    }

    @Override
    @Transactional
    public void deleteQuestion(String questionId) {

        quizRepository.deleteQuestionById(questionId);
    }

    @Override
    public QuestionResponse getQuestionById(String questionId) {

        return mapper.toResponse(
                quizRepository.findQuestionById(questionId)
                        .orElseThrow(() -> new AppException(
                                "Không tìm thấy câu hỏi",
                                HttpStatus.NOT_FOUND)));
    }

    @Override
    public List<QuestionResponse> getQuestionsByChapter(
            String chapterId) {

        return mapper.toResponseList(
                quizRepository.findQuestionsByChapterId(chapterId));
    }

package vn.id.quiz.quizcitbackend.modules.quiz.service;

import java.util.Collections;
import java.util.List;

import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import lombok.RequiredArgsConstructor;

import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Chapter;
import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Subject;
import vn.id.quiz.quizcitbackend.modules.content.repository.ContentRepository;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.dto.CreateQuizRequest;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.dto.QuizResponse;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.dto.UpdateQuizRequest;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.model.Question;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.model.Quiz;
import vn.id.quiz.quizcitbackend.modules.quiz.presentation.model.QuizQuestion;
import vn.id.quiz.quizcitbackend.modules.quiz.repository.QuizRepository;
import vn.id.quiz.quizcitbackend.shared.exceptions.AppException;

@Service
@RequiredArgsConstructor
public class QuizServiceImplement implements QuizService {

    private final QuizRepository quizRepository;
    private final ContentRepository contentRepository;
    private final QuizMapper mapper;

    @Override
    @Transactional
    public QuizResponse createQuiz(
            CreateQuizRequest request) {

        Subject subject = contentRepository.findSubjectById(
                request.getSubjectId())
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy môn học",
                        HttpStatus.NOT_FOUND));

        Chapter chapter = contentRepository.findChapterById(
                request.getChapterId())
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy chương học",
                        HttpStatus.NOT_FOUND));

        List<Question> questions = quizRepository.findQuestionsByChapterId(
                chapter.getChapterId());

        if (questions.isEmpty()) {
            throw new AppException(
                    "Không có câu hỏi",
                    HttpStatus.BAD_REQUEST);
        }

        Collections.shuffle(questions);

        List<Question> selected = questions.stream()
                .limit(request.getTotalQuestion())
                .toList();

        Quiz quiz = Quiz.create(
                subject,
                chapter,
                request.getName(),
                selected.size(),
                request.getType(),
                request.getDuration());

        quizRepository.save(quiz);

        int order = 1;

        for (Question q : selected) {

            quizRepository.save(
                    QuizQuestion.builder()
                            .quiz(quiz)
                            .question(q)
                            .indexOrder(order++)
                            .build());
        }

        return mapper.toResponse(quiz);
    }

    @Override
    @Transactional
    public void updateQuiz(
            String quizId,
            UpdateQuizRequest request) {

        Quiz quiz = quizRepository.findQuizById(
                quizId)
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy quiz",
                        HttpStatus.NOT_FOUND));

        Quiz updated = Quiz.builder()
                .quizId(quiz.getQuizId())
                .subject(quiz.getSubject())
                .chapter(quiz.getChapter())
                .name(request.getName())
                .type(quiz.getType())
                .totalQuestion(quiz.getTotalQuestion())
                .duration(request.getDuration())
                .build();

        quizRepository.save(updated);
    }

    @Override
    @Transactional
    public void deleteQuiz(String quizId) {

        quizRepository.deleteQuizById(quizId);
        quizRepository.deleteQuizQuestionsByQuizId(quizId);
    }

    @Override
    public QuizResponse getQuizById(String quizId) {

        return mapper.toResponse(
                quizRepository.findQuizById(quizId)
                        .orElseThrow(() -> new AppException(
                                "Không tìm thấy quiz",
                                HttpStatus.NOT_FOUND)));
    }

    @Override
    public List<QuizResponse> getQuizzesByChapter(
            String chapterId) {

        return mapper.toResponseList(
                quizRepository.findQuizzesByChapterId(chapterId));
    }

                                HttpStatus.NOT_FOUND));
        // Tạo danh sách các skill có trong bài
        Map<String, SkillThetaData> skillMap = new HashMap<>();

        // Duyệt đáp án
        int correctCount = 0;

        for (SubmitAnswerRequest answer : request.getAnswers()) {
            Question question = quizRepository
                    .findQuestionById(
                            answer.getQuestionId())
                    .orElseThrow();
            Option option = quizRepository
                    .findOptionById(
                            answer.getOptionId())
                    .orElseThrow();
            boolean correct = option.isCorrected();
            Response response = Response.create(
                    attempt.getAttemptId(),
                    question.getQuestionId(),
                    option.getOptionId(),
                    correct);
            quizRepository.save(response);
            if (correct) {
                correctCount++;
            }
            List<QuestionSkill> mappings = progressRepository
                    .findQuestionSkillsByQuestionId(
                            question.getQuestionId());
            for (QuestionSkill qs : mappings) {

                SkillThetaData data = skillMap.computeIfAbsent(
                        qs.getSkillId(),
                        id -> new SkillThetaData());

                data.getResponses()
                        .add(correct ? 1 : 0);

                data.getDifficulties()
                        .add(question.getDifficultyB());

                data.getWeights()
                        .add(1.0);
            }
        }
        Map<String, Double> skillThetaMap = new HashMap<>();

        for (var entry : skillMap.entrySet()) {

            String skillId = entry.getKey();

            SkillThetaData data = entry.getValue();

            double theta = aiService.estimateTheta(
                    data.getResponses(),
                    data.getDifficulties(),
                    data.getWeights());

            skillThetaMap.put(
                    skillId,
                    theta);

            UserSkillAbility ability = progressRepository
                    .findUserSkillAbility(
                            attempt.getUserId(),
                            skillId)
                    .orElse(null);

            double oldTheta = 0.0;

            if (ability == null) {

                ability = UserSkillAbility.create(
                        attempt.getUserId(),
                        skillId);

            } else {

                oldTheta = ability.getTheta();
            }

            ability.updateAbility(
                    theta,
                    0.0);

            progressRepository.save(
                    ability);

            AbilityHistory history = AbilityHistory.create(
                    attempt.getUserId(),
                    skillId,
                    theta,
                    theta - oldTheta,
                    0.0,
                    attempt.getAttemptId());

            progressRepository.save(
                    history);
        }
        double overallTheta = skillThetaMap.values()
                .stream()
                .mapToDouble(Double::doubleValue)
                .average()
                .orElse(0.0);
        double rawScore = (double) correctCount
                / attempt.getTotalQuestion();
        attempt.complete(
                rawScore,
                overallTheta);

        quizRepository.save(
                attempt);
        return mapper.toQuizResultResponse(
                attempt);
    }

    @Override
    @Transactional(readOnly = true)
    public QuizResultResponse getResult(
            String attemptId) {

        Attempt attempt = quizRepository
                .findAttemptById(attemptId)
                .orElseThrow(() -> new AppException(
                        "Không tìm thấy kết quả bài thi",
                        HttpStatus.NOT_FOUND));

        List<AbilityHistory> histories = progressRepository.findAbilityHistoriesByAttemptId(
                attemptId);

        List<SkillResultResponse> skillResults = histories.stream()
                .map(history -> SkillResultResponse.builder()
                        .skillId(history.getSkillId())
                        .theta(history.getTheta())
                        .standardError(
                                history.getStandardError())
                        .build())
                .toList();

        return QuizResultResponse.builder()
                .attemptId(attempt.getAttemptId())
                .quizId(attempt.getQuizId())
                .rawScore(attempt.getRawScore())
                .totalQuestion(attempt.getTotalQuestion())
                .overallTheta(attempt.getOverallTheta())
                .skillResults(skillResults)
                .build();
    }

    @Override
    @Transactional(readOnly = true)
    public List<QuizResultResponse> getUserResults(
            String userId) {

        List<Attempt> attempts = quizRepository.findAttemptsByUserId(
                userId);

        return mapper.toQuizResultResponseList(
                attempts);
    }

package vn.id.quiz.quizcitbackend.modules.content.service;

import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Chapter;
import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Subject;
import vn.id.quiz.quizcitbackend.modules.content.presentation.model.Theory;
import vn.id.quiz.quizcitbackend.modules.content.presentation.dto.*;
import vn.id.quiz.quizcitbackend.modules.content.repository.ContentRepository;
import vn.id.quiz.quizcitbackend.shared.exceptions.AppException;

import java.util.List;

@Service
@RequiredArgsConstructor
public class ContentServiceImplement implements ContentService {
    private final ContentRepository contentRepository;
    private final ContentMapper mapper;

    @Override
    @Transactional(readOnly = true)
    public List<SubjectResponse> getAllSubject(){
        List<Subject> subjects = contentRepository.findAllSubject();
        return mapper.toSubjectResponseList(subjects);
    }
    @Override
    @Transactional(readOnly = true)
    public SubjectResponse getSubjectById(String subjectId){

        Subject subject = contentRepository.findSubjectById(subjectId)
                .orElseThrow(()-> new AppException("Không tìm thấy học phần được yêu cầu", HttpStatus.NOT_FOUND));
        return mapper.toSubjectResponse(subject);
    }
    @Override
    @Transactional(readOnly = true)
    public ChapterResponse getChapterById(String chapterId){
        Chapter chapter = contentRepository.findChapterById(chapterId)
                .orElseThrow(()-> new AppException("Không tìm thấy chương được yêu cầu", HttpStatus.NOT_FOUND));
        return mapper.toChapterResponse(chapter);
    }
    @Override
    @Transactional(readOnly = true)
    public TheoryResponse getTheoryById(String theoryId){
        Theory theory = contentRepository.findTheoryById(theoryId)
                .orElseThrow(()-> new AppException("Không tìm thấy nội dung lý thuyết được yêu cầu", HttpStatus.NOT_FOUND));
        return mapper.toTheoryResponse(theory);
    }

    @Override
    @Transactional
    public SubjectResponse createSubject(CreateSubjectRequest request){

        if(contentRepository.existsByCourseCode(request.courseCode())){
            throw new AppException("Mã học phần đã tồn tại.", HttpStatus.CONFLICT);
        }
        Subject subject = Subject.create(request.courseCode(), request.name(), request.description());
        return mapper.toSubjectResponse(contentRepository.save(subject));
    }

    @Override
    @Transactional
    public ChapterResponse addChapter(String subjectId, CreateChapterRequest request){
        if(!contentRepository.existsBySubjectId(subjectId)){
            throw new AppException("Không tìm thấy học phần được yêu cầu", HttpStatus.NOT_FOUND);
        }
        Chapter response = Chapter.create(subjectId, request.name());
        contentRepository.save(response);
        return mapper.toChapterResponse(response);
    }
    @Override
    @Transactional
    public TheoryResponse addTheory(String chapterId, CreateTheoryRequest request){
        if(!contentRepository.existsByChapterId(chapterId)){
            throw new AppException("Không tìm thấy chương được yêu cầu", HttpStatus.NOT_FOUND);
        }

        Theory response = Theory.create(chapterId, request.content());
        contentRepository.save(response);
        return mapper.toTheoryResponse(response);
    }


    @Override
    @Transactional(readOnly = true)
    public List<ChapterResponse> getAllChapters(String subjectId){
        if(!contentRepository.existsBySubjectId(subjectId)){
            throw new AppException("Không tìm thấy học phần được yêu cầu", HttpStatus.NOT_FOUND);
        }
        List<Chapter> chapters = contentRepository.findAllChapter(subjectId);
        return mapper.toChapterResponseList(chapters);
    }

    @Override
    @Transactional(readOnly = true)
    public List<TheoryResponse> getAllTheories(String chapterId){
        if(!contentRepository.existsByChapterId(chapterId)){
            throw new AppException("Không tìm thấy chương được yêu cầu", HttpStatus.NOT_FOUND);
        }
        List<Theory> theories = contentRepository.findAllTheory(chapterId);
        return mapper.toTheoryResponseList(theories);
    }
    @Transactional
    @Override
    public void updateSubject(String subjectId, UpdateSubjectRequest request){
        Subject subject = contentRepository.findSubjectById(subjectId)
                .orElseThrow(()-> new AppException("Không tìm thấy học phần được yêu cầu", HttpStatus.NOT_FOUND));
        subject.updateSubject(request.courseCode(), request.name(), request.description());
        contentRepository.save(subject);
    }
    @Transactional
    @Override
    public void updateChapter(String chapterId, UpdateChapterRequest request){
        Chapter chapter = contentRepository.findChapterById(chapterId)
                        .orElseThrow(()->new AppException("Không tìm thấy chương được yêu cầu", HttpStatus.NOT_FOUND));

        chapter.updateChapter(request.name());
        contentRepository.save(chapter);
    }
    @Transactional
    @Override
    public void updateTheory(String theoryId, UpdateTheoryRequest request){
        Theory theory = contentRepository.findTheoryById(theoryId)
                .orElseThrow(()-> new AppException("Không tìm thấy nội dung lý thuyết được yêu cầu", HttpStatus.NOT_FOUND));
        theory.updateTheory(request.content());
        contentRepository.save(theory);
    }

    @Override
    @Transactional
    public void deleteSubject(String subjectId){
        contentRepository.deleteSubjectById(subjectId);
    }
    @Override
    @Transactional
    public void deleteChapter(String chapterId) {
        contentRepository.deleteChapterById(chapterId);
    }

    @Override
    @Transactional
    public void deleteTheory(String theoryId) {
        contentRepository.deleteTheoryById(theoryId);
    }
}
